<script setup lang="ts">
import { can } from '@layouts/plugins/casl'
import type { SendSuccessInterface } from '@/types/sistema/types'
import type DataTable from '@/components/form-elements/DataTableComponent.vue'

definePageMeta({
  middleware: 'permissions',
  action: 'listar {{ nombreModeloPermiso }}',
  subject: '{{ model }}',
})

const { deleted } = useClienteRequest()
const { showToastSuccess, preguntaEliminar } = useToast()
const { paginaEspera } = useCargandoPagina()

const headers = {{ headers }}

const dataTable = ref<InstanceType<typeof DataTable> | null>(null)

const deleteItem = async (id: number) => {
  const confirm = await preguntaEliminar('¿Desea Eliminar {{ model }}?')
  if (!confirm)
    return
  paginaEspera.value = true
  try {
    const respuesta = await deleted<SendSuccessInterface>(`{{ url }}/${id}`)

    showToastSuccess(respuesta.message)
    if (dataTable.value)
      dataTable.value?.getItems()
  }
  catch (error: any) {
    manejaError(error)
  }
  finally {
    paginaEspera.value = false
  }
}
</script>

<template>
  <div class="d-flex flex-wrap justify-end justify-sm-space-between gap-y-4 gap-x-6 mb-4">
    <h1 v-text="'Listado de {{ modelPlural }}'"/>
    <VBtn
      v-if="can('crear {{ nombreModeloPermiso }}', '{{ model }}')"
      class="ml-auto"
      to="/{{ directory }}/create"
    >
      <VIcon class="tabler-plus mr-1"/>
      Nuev@ {{ model }}
    </VBtn>
  </div>

  <DataTableComponent
    ref="dataTable"
    :columnas="headers"
    endpoint="{{ url }}"
    :cantidad-por-pagina="5"
    :cantidad-por-pagina-opciones="[10, 20, 30]"
    :botones="['xlsx', 'pdf', 'csv', 'reiniciar']"
  >
    <template #item.Acciones="{ item }">
      <VBtn
        v-if="can('ver {{ nombreModeloPermiso }}', '{{ model }}')"
        icon="tabler-eye"
        variant="tonal"
        v-tooltip="'Ver'"
        color="info"
        :to="`/{{ directory }}/show/${item.id}`"
        class="mr-1"
      />
      <VBtn
        v-if="can('editar {{ nombreModeloPermiso }}', '{{ model }}')"
        icon="tabler-edit"
        variant="tonal"
        v-tooltip="'Editar'"
        color="warning"
        :to="`/{{ directory }}/edit/${item.id}`"
        class="mr-1"
      />
      <VBtn
        v-if="can('eliminar {{ nombreModeloPermiso }}', '{{ model }}')"
        icon="tabler-trash"
        variant="tonal"
        v-tooltip="'Eliminar'"
        color="error"
        @click="deleteItem(item.id)"
      />
    </template>
  </DataTableComponent>
</template>
